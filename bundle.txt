package com.example.fluidnotif

import android.content.Context
import android.graphics.Canvas
import android.graphics.LinearGradient
import android.graphics.Paint
import android.graphics.RectF
import android.graphics.RenderEffect
import android.graphics.Shader
import android.graphics.Typeface
import android.text.TextUtils
import android.view.Gravity
import android.widget.FrameLayout
import android.widget.ImageView
import android.widget.LinearLayout
import android.widget.TextView
import androidx.dynamicanimation.animation.FloatPropertyCompat

/**
 * Glass notification card. [progress] 0 = small capsule at the camera,
 * 1 = fully expanded card. Driven by a spring animation.
 */
class GlassCardView(context: Context, d: NotifData, private val st: Style) : FrameLayout(context) {

    companion object {
        val PROGRESS = object : FloatPropertyCompat<GlassCardView>("progress") {
            override fun getValue(o: GlassCardView) = o.progress
            override fun setValue(o: GlassCardView, value: Float) { o.progress = value }
        }
    }

    /** Capsule rect (where the camera is) in this view's local coordinates. */
    var startRect = RectF()

    var progress = 0f
        set(v) {
            field = v
            applyProgress()
            invalidate()
        }

    private val fill = Paint(Paint.ANTI_ALIAS_FLAG)
    private val stroke = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        style = Paint.Style.STROKE
        strokeWidth = context.dpF(st.edgeWDp)
    }
    private val glow = Paint(Paint.ANTI_ALIAS_FLAG).apply {
        style = Paint.Style.STROKE
        strokeWidth = context.dpF(st.edgeWDp * 4f)
        alpha = st.glowAlpha
    }
    private val rect = RectF()
    private val content = LinearLayout(context)

    private fun withA(c: Int, a: Int): Int = (c and 0x00FFFFFF) or (a.coerceIn(0, 255) shl 24)

    init {
        setWillNotDraw(false)

        val cHeader = if (st.dark) 0xFFB0B0BA.toInt() else 0xFF6B6B73.toInt()
        val cTitle = if (st.dark) 0xFFF4F4F8.toInt() else 0xFF121216.toInt()
        val cBody = if (st.dark) 0xFFD0D0D8.toInt() else 0xFF4A4A52.toInt()
        val s = st.textScale

        content.orientation = LinearLayout.HORIZONTAL
        content.gravity = Gravity.CENTER_VERTICAL
        content.setPadding(dpI(16f), dpI(14f), dpI(16f), dpI(14f))

        val icon = ImageView(context).apply {
            setImageDrawable(d.icon())
            scaleType = ImageView.ScaleType.FIT_CENTER
        }
        val isz = dpI(st.iconDp)
        content.addView(icon, LinearLayout.LayoutParams(isz, isz))

        val col = LinearLayout(context).apply { orientation = LinearLayout.VERTICAL }
        col.addView(TextView(context).apply {
            text = "${d.appName} · now"
            textSize = 12f * s
            setTextColor(cHeader)
            maxLines = 1
            ellipsize = TextUtils.TruncateAt.END
        })
        col.addView(TextView(context).apply {
            text = d.title
            textSize = 15.5f * s
            setTextColor(cTitle)
            setTypeface(typeface, Typeface.BOLD)
            maxLines = 1
            ellipsize = TextUtils.TruncateAt.END
        })
        col.addView(TextView(context).apply {
            text = if (st.hide) "Message hidden" else d.text
            textSize = 14f * s
            setTextColor(cBody)
            maxLines = 2
            ellipsize = TextUtils.TruncateAt.END
        })
        val lp = LinearLayout.LayoutParams(0, WRAP, 1f)
        lp.marginStart = dpI(12f)
        content.addView(col, lp)

        addView(content, LayoutParams(MATCH, WRAP))
        applyProgress()
    }

    private fun applyProgress() {
        // Content fades in and un-blurs during the second half of the expansion.
        val a = ((progress - 0.35f) / 0.5f).coerceIn(0f, 1f)
        content.alpha = a
        val r = (1f - a) * dpF(st.textBlurDp)
        content.setRenderEffect(
            if (r > 0.5f) RenderEffect.createBlurEffect(r, r, Shader.TileMode.CLAMP) else null
        )
    }

    override fun onDraw(canvas: Canvas) {
        val p = progress.coerceAtLeast(0f)
        val w = width.toFloat()
        val h = height.toFloat()
        rect.set(
            lerp(startRect.left, 0f, p),
            lerp(startRect.top, 0f, p),
            lerp(startRect.right, w, p),
            lerp(startRect.bottom, h, p)
        )
        val pad = if (st.edgeOn) stroke.strokeWidth / 2f else 0f
        rect.inset(pad, pad)
        val radius = lerp(startRect.height() / 2f, dpF(st.cornerDp), p.coerceAtMost(1f))

        // Milky translucent glass body
        val top = if (st.dark) 0x2E2E36 else 0xFFFFFF
        val bot = if (st.dark) 0x17171C else 0xECECF2
        fill.shader = LinearGradient(
            rect.left, rect.top, rect.right, rect.bottom,
            withA(top, st.glassAlpha), withA(bot, (st.glassAlpha * 0.82f).toInt()),
            Shader.TileMode.CLAMP
        )
        canvas.drawRoundRect(rect, radius, radius, fill)

        if (st.edgeOn) {
            val edge = LinearGradient(
                rect.left, rect.top, rect.right, rect.bottom,
                st.c1, st.c2, Shader.TileMode.CLAMP
            )
            glow.shader = edge
            stroke.shader = edge
            canvas.drawRoundRect(rect, radius, radius, glow)
            canvas.drawRoundRect(rect, radius, radius, stroke)
        }
    }
}

package com.example.fluidnotif

import android.content.Context
import android.graphics.RectF
import android.graphics.Typeface
import android.graphics.drawable.Drawable
import android.graphics.drawable.GradientDrawable
import android.os.Handler
import android.os.Looper
import android.view.Gravity
import android.view.HapticFeedbackConstants
import android.view.MotionEvent
import android.view.animation.OvershootInterpolator
import android.widget.EditText
import android.widget.LinearLayout
import android.widget.TextView
import androidx.dynamicanimation.animation.SpringAnimation
import androidx.dynamicanimation.animation.SpringForce
import kotlin.math.abs

/**
 * The whole notification UI: glass card + action pills, with open/close animations.
 * Used both inside the system overlay window and inside the settings preview.
 */
class NotifPanel(
    private val ctx: Context,
    private val d: NotifData,
    private val st: Style,
    startRect: RectF,
    cardTop: Int
) : LinearLayout(ctx) {

    val card = GlassCardView(ctx, d, st)
    private val row = LinearLayout(ctx)
    private val handler = Handler(Looper.getMainLooper())
    private var autoClose: Runnable? = null
    private var closing = false
    private val fg = if (st.dark) 0xFFF4F4F8.toInt() else 0xFF121216.toInt()
    private val hasButtons = st.buttons && (st.btnReply || st.btnRead)

    var onClosed: (() -> Unit)? = null
    var onNeedFocus: ((EditText) -> Unit)? = null

    init {
        orientation = VERTICAL
        card.startRect = startRect
        card.progress = 0f
        addView(card, LayoutParams(MATCH, WRAP).apply { topMargin = cardTop.coerceAtLeast(0) })

        row.orientation = HORIZONTAL
        row.gravity = Gravity.CENTER_HORIZONTAL
        if (hasButtons) {
            addView(row, LayoutParams(MATCH, WRAP).apply { topMargin = dpI(8f) })
            buildButtons()
            row.alpha = 0f
            row.translationY = -dpF(18f)
            row.scaleX = 0.9f
            row.scaleY = 0.9f
        }

        var downY = 0f
        card.setOnTouchListener { _, e ->
            when (e.actionMasked) {
                MotionEvent.ACTION_DOWN -> { downY = e.rawY; true }
                MotionEvent.ACTION_UP -> {
                    val dy = e.rawY - downY
                    if (dy < -dpF(28f)) {
                        close()
                    } else if (abs(dy) < dpF(8f)) {
                        d.onOpen()
                        close()
                    }
                    true
                }
                else -> true
            }
        }
    }

    fun open(haptic: Boolean = st.haptic) {
        closing = false
        if (haptic) card.performHapticFeedback(HapticFeedbackConstants.CONFIRM)
        SpringAnimation(card, GlassCardView.PROGRESS, 1f).apply {
            spring.stiffness = st.openStiffness
            spring.dampingRatio = st.openDamping
            start()
        }
        if (hasButtons) {
            row.animate()
                .alpha(1f).translationY(0f).scaleX(1f).scaleY(1f)
                .setStartDelay(st.btnDelay).setDuration(320)
                .setInterpolator(OvershootInterpolator(1.4f))
                .start()
        }
        scheduleAutoClose()
    }

    fun close() {
        if (closing) return
        closing = true
        cancelAuto()
        if (hasButtons) {
            row.animate().alpha(0f).translationY(-dpF(10f)).setStartDelay(0).setDuration(160).start()
        }
        SpringAnimation(card, GlassCardView.PROGRESS, 0f).apply {
            spring.stiffness = st.closeStiffness
            spring.dampingRatio = SpringForce.DAMPING_RATIO_NO_BOUNCY
            addEndListener { _, _, _, _ -> onClosed?.invoke() }
            start()
        }
    }

    fun release() {
        cancelAuto()
        handler.removeCallbacksAndMessages(null)
        row.animate().cancel()
    }

    private fun scheduleAutoClose() {
        cancelAuto()
        autoClose = Runnable { close() }.also { handler.postDelayed(it, st.stayMs) }
    }

    private fun cancelAuto() {
        autoClose?.let { handler.removeCallbacks(it) }
        autoClose = null
    }

    private fun buildButtons() {
        row.removeAllViews()
        if (st.btnReply) {
            row.addView(pill("Reply", d.icon()) {
                if (d.canReply) showReply() else { d.onOpen(); close() }
            })
        }
        if (st.btnRead) {
            val check = ctx.getDrawable(R.drawable.ic_check)?.mutate()?.also { it.setTint(fg) }
            row.addView(pill("Mark as read", check) {
                d.onRead()
                close()
            }, LayoutParams(WRAP, WRAP).apply { marginStart = if (st.btnReply) dpI(8f) else 0 })
        }
    }

    private fun showReply() {
        cancelAuto()
        row.removeAllViews()
        val input = EditText(ctx).apply {
            hint = "Reply…"
            textSize = 15f
            setSingleLine()
            setTextColor(fg)
            setHintTextColor(0xFF8A8A92.toInt())
            setPadding(dpI(20f), 0, dpI(20f), 0)
            background = glassBg()
        }
        row.addView(input, LayoutParams(0, dpI(48f), 1f))
        row.addView(pill("Send", null) {
            val text = input.text.toString().trim()
            if (text.isNotEmpty()) d.onReply(text)
            close()
        }, LayoutParams(WRAP, WRAP).apply { marginStart = dpI(8f) })
        row.alpha = 1f
        row.translationY = 0f
        row.scaleX = 1f
        row.scaleY = 1f
        onNeedFocus?.invoke(input)
    }

    private fun glassBg() = GradientDrawable().apply {
        cornerRadius = dpF(24f)
        setColor(if (st.dark) 0xE62A2A30.toInt() else 0xEBFFFFFF.toInt())
        setStroke(dpI(1f), 0x55FFFFFF)
    }

    private fun pill(label: String, icon: Drawable?, onClick: () -> Unit): TextView {
        return TextView(ctx).apply {
            text = label
            textSize = 15f * st.textScale
            setTextColor(fg)
            setTypeface(typeface, Typeface.BOLD)
            gravity = Gravity.CENTER_VERTICAL
            minHeight = dpI(48f)
            setPadding(dpI(if (icon != null) 14f else 22f), 0, dpI(22f), 0)
            compoundDrawablePadding = dpI(8f)
            icon?.let {
                it.setBounds(0, 0, dpI(26f), dpI(26f))
                setCompoundDrawablesRelative(it, null, null, null)
            }
            background = glassBg()
            setOnClickListener { onClick() }
        }
    }
}

package com.example.fluidnotif

import android.app.Notification
import android.app.RemoteInput
import android.content.Context
import android.content.Intent
import android.os.Bundle
import android.service.notification.NotificationListenerService
import android.service.notification.StatusBarNotification

class NotifService : NotificationListenerService() {

    private val skipCategories = setOf(
        Notification.CATEGORY_TRANSPORT, Notification.CATEGORY_PROGRESS,
        Notification.CATEGORY_SERVICE, Notification.CATEGORY_SYSTEM,
        Notification.CATEGORY_STATUS
    )

    override fun onNotificationPosted(sbn: StatusBarNotification) {
        val n = sbn.notification
        val st = Style(getSharedPreferences("fluid", Context.MODE_PRIVATE))

        if (sbn.packageName in st.blocked) return
        if (n.flags and Notification.FLAG_GROUP_SUMMARY != 0) return
        if (st.skipOngoing && (sbn.isOngoing || n.category in skipCategories)) return

        val ex = n.extras
        val title = ex.getCharSequence(Notification.EXTRA_TITLE)?.toString().orEmpty()
        val text = (ex.getCharSequence(Notification.EXTRA_BIG_TEXT)
            ?: ex.getCharSequence(Notification.EXTRA_TEXT))?.toString().orEmpty()
        if (title.isEmpty() && text.isEmpty()) return

        val pm = packageManager
        val pkg = sbn.packageName
        val appName = try {
            pm.getApplicationLabel(pm.getApplicationInfo(pkg, 0)).toString()
        } catch (e: Exception) { pkg }

        val acts = n.actions.orEmpty()
        val replyAct = acts.firstOrNull { !it.remoteInputs.isNullOrEmpty() }
        val readAct = acts.firstOrNull {
            it.semanticAction == Notification.Action.SEMANTIC_ACTION_MARK_AS_READ ||
                it.title?.contains("read", ignoreCase = true) == true
        }
        val ri = replyAct?.remoteInputs?.firstOrNull()

        val data = NotifData(
            appName = appName,
            icon = { try { pm.getApplicationIcon(pkg) } catch (e: Exception) { null } },
            title = title,
            text = text,
            canReply = ri != null,
            onOpen = {
                try { n.contentIntent?.send() } catch (_: Exception) {}
                cancelNotification(sbn.key)
            },
            onRead = {
                try { readAct?.actionIntent?.send() } catch (_: Exception) {}
                cancelNotification(sbn.key)
            },
            onReply = { reply ->
                try {
                    if (ri != null) {
                        val i = Intent()
                        val b = Bundle().apply { putCharSequence(ri.resultKey, reply) }
                        RemoteInput.addResultsToIntent(arrayOf(ri), i, b)
                        replyAct.actionIntent.send(applicationContext, 0, i)
                    }
                } catch (_: Exception) {}
                cancelNotification(sbn.key)
            }
        )
        OverlayController.show(applicationContext, data)
    }
}

/** Puts a [NotifPanel] into a system overlay window anchored at the camera. */
object OverlayController {

    private var wm: WindowManager? = null
    private var panel: NotifPanel? = null

    fun show(base: Context, d: NotifData) {
        if (!Settings.canDrawOverlays(base)) return
        removeNow()

        val st = Style(base.getSharedPreferences("fluid", Context.MODE_PRIVATE))
        val ctx = ContextThemeWrapper(base, android.R.style.Theme_DeviceDefault_Light_NoActionBar)
        val manager = base.getSystemService(WindowManager::class.java)
        wm = manager

        // Where is the camera?
        val metrics = manager.currentWindowMetrics
        val screenW = metrics.bounds.width()
        val insets = metrics.windowInsets
        val statusTop = insets.getInsets(WindowInsets.Type.statusBars()).top
        val cut = insets.displayCutout?.boundingRects?.firstOrNull { it.top <= 0 }
        val cam = if (cut != null) RectF(cut) else RectF(
            screenW / 2f - ctx.dpF(14f), ctx.dpF(8f),
            screenW / 2f + ctx.dpF(14f), ctx.dpF(36f)
        )
        cam.inset(-ctx.dpF(3f), -ctx.dpF(3f))

        val margin = ctx.dpI(st.marginDp)
        val cardTop = (maxOf(cam.bottom.toInt(), statusTop) + ctx.dpI(st.topGapDp)).coerceAtLeast(0)
        val start = RectF(
            cam.left - margin, cam.top - cardTop,
            cam.right - margin, cam.bottom - cardTop
        )

        val p = NotifPanel(ctx, d, st, start, cardTop)

        val lp = WindowManager.LayoutParams(
            screenW - margin * 2,
            WRAP,
            WindowManager.LayoutParams.TYPE_APPLICATION_OVERLAY,
            WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE or
                WindowManager.LayoutParams.FLAG_NOT_TOUCH_MODAL or
                WindowManager.LayoutParams.FLAG_LAYOUT_IN_SCREEN or
                WindowManager.LayoutParams.FLAG_LAYOUT_NO_LIMITS,
            PixelFormat.TRANSLUCENT
        ).apply {
            gravity = Gravity.TOP or Gravity.CENTER_HORIZONTAL
            y = 0
            layoutInDisplayCutoutMode = WindowManager.LayoutParams.LAYOUT_IN_DISPLAY_CUTOUT_MODE_ALWAYS
            if (st.blur) {
                // Experimental: blurs the whole window rectangle, not only the rounded card.
                flags = flags or WindowManager.LayoutParams.FLAG_BLUR_BEHIND
                blurBehindRadius = ctx.dpI(st.blurDp)
            }
        }

        p.onClosed = { if (panel === p) removeNow() }
        p.onNeedFocus = { input ->
            // Make the overlay focusable so the keyboard can appear.
            lp.flags = lp.flags and WindowManager.LayoutParams.FLAG_NOT_FOCUSABLE.inv()
            lp.softInputMode = WindowManager.LayoutParams.SOFT_INPUT_ADJUST_NOTHING
            try { manager.updateViewLayout(p, lp) } catch (_: Exception) {}
            input.requestFocus()
            input.postDelayed({
                ctx.getSystemService(InputMethodManager::class.java)
                    .showSoftInput(input, InputMethodManager.SHOW_IMPLICIT)
            }, 150)
        }

        manager.addView(p, lp)
        panel = p
        p.open()
    }

    fun dismiss() {
        panel?.close()
    }

    private fun removeNow() {
        panel?.let {
            it.release()
            try { wm?.removeViewImmediate(it) } catch (_: Exception) {}
        }
        panel = null
    }
}
